//제미나이 개선 코드 한줄 한줄 이해하기 

import java.util.Scanner;
import java.util.ArrayList;

// 1. 규격 선언 (인터페이스): "모든 제품은 구매 기능이 있어야 한다"
interface Function {
    void buy(int money); // 추상 메소드
}

// 2. 실체 구현 (클래스): "자판기 제품의 설계도"
class VendingMachine implements Function {
    String productName;
    int productMoney;
    int quantity;

    VendingMachine(String productName, int productMoney) {
        this.productName = productName;
        this.productMoney = productMoney;
        this.quantity = 5; // 재고는 기본 5개로 세팅
    }

    @Override
    public void buy(int money) {
        // 재고 부족 체크
        if (quantity <= 0) {
            System.out.println(">>> [알림] " + productName + "은(는) 품절되었습니다.");
            return;
        }

        // 돈 부족 체크 및 구매 처리
        if (money >= productMoney) {
            this.quantity--; // 재고 감소
            System.out.println(">>> [성공] " + productName + " 구매 완료!");
            System.out.println(">>> [잔돈] " + (money - productMoney) + "원이 반환됩니다.");
        } else {
            System.out.println(">>> [실패] 잔액이 부족합니다. (" + (productMoney - money) + "원 더 필요)");
        }
    }
}

// 3. 운영 (메인): "자판기를 돌리는 시스템"
public class Main {
    public static void main(String[] args) {
        // 제품 데이터베이스 구축 (ArrayList 활용)
        ArrayList<VendingMachine> list = new ArrayList<>();
        list.add(new VendingMachine("포카리스웨트", 1000));
        list.add(new VendingMachine("코카콜라", 2000));
        list.add(new VendingMachine("데자와", 3000));
        list.add(new VendingMachine("조지아라떼", 4000));

        Scanner sc = new Scanner(System.in);

        // 무한 루프: 종료를 선택하기 전까지 자판기는 계속 가동됨
        while (true) {
            System.out.println("\n======= 상어의 자판기 (Ver 1.0) =======");
            for (int i = 0; i < list.size(); i++) {
                // 리스트 인덱스(0,1,2,3)에 1을 더해 사용자용 번호(1,2,3,4) 생성
                System.out.println((i + 1) + ". " + list.get(i).productName + " | 가격: " + list.get(i).productMoney + "원 | 재고: " + list.get(i).quantity);
            }
            System.out.println("0. 프로그램 종료");
            System.out.print("선택(번호): ");
            
            int choice = sc.nextInt();

            if (choice == 0) {
                System.out.println("자판기를 종료합니다. 오늘도 고생하셨습니다!");
                break; // while문 탈출
            }

            // 트러블슈팅: 잘못된 인덱스 입력 방어 (1~4번 외의 번호)
            if (choice < 1 || choice > list.size()) {
                System.out.println("잘못된 번호입니다. 다시 선택하십시오.");
                continue; // 루프 처음으로 이동
            }

            System.out.print("투입 금액: ");
            int inputMoney = sc.nextInt();

            // 핵심: 선택한 번호(choice-1)의 객체만 불러내서 buy 실행!
            list.get(choice - 1).buy(inputMoney);
        }
        sc.close();
    }
}





여기서 얻게된 개념




1. 익명객체 (Anonymous Object)

위에 어레이리스트에 그냥 add로 넣으면서 생성자로 변수 생성과동시에 집어넣는것보이나

이렇게 참조변수 /참조변수명 없이 태어나는 객체들을 익명객체라한다

이 경우에 참조변수가 따로없기에 

어레이리스트만으로 조종이 가능하다

하지만 이 경우 코드가 매우 깔끔해진다.
굳이 개별 일일이 참조변수명 선언해줄필요가없기에

그냥 저것들은 전부다 배열의 0번째놈 1번째놈 2번째놈 이렇게 관리해줄수있게되는것이다

개별 관리 x 단체 관리 

그러면서도 list.get 사용해서 
배열속 개별 객체를 불러와서 개별 메소드 선언해주는것은 참 고무적이다. 
관리의 단일화로 간단하게 만들어주고 
외부라이브러리인 어레이리스트속 내장메소드 get을 사용해서 
어레이리스트속 개별객체를 호출후 메소드 실행 - 개별관리 

2.while 무한루프문의 실용적 사용 

while은 조건식안이 참일경우 계속 돌아가는 루프문이다

그걸 알기에 항상 작동하게 해주고 

에러가 생긴 경우를 if로 묶어주고 그 경우에 break로 탈출시켜버리면 되는것이다.
