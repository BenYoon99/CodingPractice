import java.util.Scanner;
import java.lang.*;
import java.io.*;

/*. 요구사항
인터페이스 Bob: void eat(); 추상 메서드를 가집니다.

부모 클래스 Mom: Bob을 구현(implements)하며, String name 필드를 가집니다. eat() 호출 시 
**"{name}이(가) 정성스럽게 차린 밥을 먹습니다."**를 출력합니다.

자식 클래스 Shark: Mom을 상속(extends)받습니다. eat()를 오버라이딩하여 
**"{name}이(가) 독기를 품고 단백질 식단을 먹습니다!"**를 출력합니다.

메인 함수 (Main):

Scanner를 사용하여 Mom 객체와 Shark 객체의 name을 각각 입력받습니다.

동적 할당: Bob 타입의 배열(크기 2)을 생성합니다.

동적 바인딩: 배열의 0번에는 Mom 객체를, 1번에는 Shark 객체를 넣고 반복문을 통해 각각의 eat()를 호출하십시오.*/


interface Bob {

    void eat();
    
}

class Mom implements Bob{

    String name;

    @Override

    public void eat(){
        System.out.println(name+"이(가) 정성스럽게 차린 밥을 먹습니다");
    }
    
}
class Shark extends Mom{

    @Override

    public void eat(){
        System.out.println(name+"이(가) 독하게 단백질 식단을 먹습니다!");
    }
}
class Main {
    public static void main(String[] args) {
        
        Mom shark = new Shark();
        Mom momShark = new Mom();
        Scanner sc = new Scanner(System.in);
        System.out.println("이름을 입력하세요:");
        shark.name =sc.next();
        
        System.out.println("엄마 이름을 입력하세요:");
        shark.name=sc.next();

        Mom [] sharks = new Mom [2];
        sharks [0] = shark;
        sharks [1] = momShark;
        
        for (Mom a : sharks){
            System.out.println(a.name);
        }
        
        
        System.out.println("Hello world!");
    }
}
