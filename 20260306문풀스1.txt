import java.util.*;
import java.lang.*;
import java.io.*;

/*🔱 6. 인터페이스와 다중 구현 (Interface)
- 문제: 1. Playable 인터페이스를 만드십시오. (void play() 메서드 포함)
2. Stoppable 인터페이스를 만드십시오. (void stop() 메서드 포함)
3. MusicPlayer 클래스를 만들고, 이 두 인터페이스를 동시에 구현하십시오.
4. 각 메서드에서는 "음악 재생", "음악 정지"를 출력하게 하십시오.

장인의 Tip: 자바에서 상속(extends)은 하나만 가능하지만, 인터페이스 구현(implements)은 콤마(,)를 찍고 무한대로 가능합니다! 👊*/

interface Playable {
    void play();
}

interface Stoppable  {

    void stop();
}

class MusicPlayer implements Playable,Stoppable {

    @Override 

    public void play(){
        System.out.println("음악 재생");
    }


    @Override 

    public void stop(){
        System.out.println("음악 정지");
    }
    
}



class Main {
    public static void main(String[] args) {
        MusicPlayer mp = new MusicPlayer();
        mp.play();
        mp.stop();
        
    }
}
