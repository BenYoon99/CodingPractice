import java.util.*;
import java.lang.*;
import java.io.*;

/*🔱 문제 2: OR (||) 실전 사격 훈련
[문제] 이번엔 변수를 두 개 쓰겠습니다. 
토익 점수가 980점 이상이거나, 학점이 4.0 이상이면 "지거국 합격권입니다"를 출력하고,
둘 다 아니면 "재도전하십시오"를 출력하는 코드를 작성해 보세요.*/

class Main {
    public static void main(String[] args) {
        int toeicScore = 920;
        double score = 3.0;


        if (toeicScore>=980 || score>=4.0){
            System.out.println("지거국 합격권입니다");
            
        }
        else{
            System.out.println("재도전하십시오");
        }
    }
}
