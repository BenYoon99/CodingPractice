class Main {

                    SkillExcute[i].execute();

                {
                    SkillExcute[i].execute();
                }

    }

class Skill {
    void execute() {
        System.out.println("기본 공격 시전");
    }
}

class FireBall extends Skill {
    @Override 

    void execute (){
        System.out.println("화염구가 폭발하며 80의 데미지를 줍니다!");
        
    }
}
class Heal extends Skill{
    @Override

    void execute() {
        System.out.println("빛의 권능으로 아군의 체력을 50 회복합니다");
    }
}
