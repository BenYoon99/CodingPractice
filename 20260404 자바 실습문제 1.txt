제미나이 수정본


interface Attackable {
    void attack(); // 몸통 없이 선언만 합니다.
}

abstract class Character {
    String name;
    int level;

    // 생성자: 반환 타입(void 등)을 적지 않습니다.
    Character(String name) {
        this.name = name;
    }

    // 모든 자식이 가져야 할 공통 기능
    void move() {
        System.out.println(name + "이(가) 이동합니다.");
    }
}

class Warrior extends Character implements Attackable {
    Warrior(String name) {
        super(name); // 부모 생성자 호출
    }

    @Override
    public void move() {
        System.out.println(name + "이 전사답게 멋있게 이동합니다.");
    }

    @Override
    public void attack() {
        System.out.println("공격 기모링 (칼 휘두르기)");
    }
}

class Mage extends Character implements Attackable {
    Mage(String name) {
        super(name);
    }

    @Override
    public void attack() {
        System.out.println("마법진을 사용합니다.");
    }

    @Override
    public void move() {
        System.out.println(name + "이 마법사답게 이동합니다.");
    }
}

class Main {
    public static void main(String[] args) {
        // 부모 타입 배열에 자식 객체를 담는 '다형성'
        Character[] arr = new Character[2];
        arr[0] = new Mage("제이나");
        arr[1] = new Warrior("바리안");

        for (Character c : arr) {
            c.move();
            
            // Character에는 attack이 없으므로, Attackable인지 확인 후 형변환 호출
            if (c instanceof Attackable) {
                ((Attackable) c).attack();
            }
        }
    }
}
