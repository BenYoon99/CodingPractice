import java.util.*;
import java.lang.*;
import java.io.*;

/*🔱 제 3관문: [초급] 논리 연산자(||)와 조건문
이제 머리를 조금 더 써서 논리 연산자의 우선순위와 조건을 다루는 훈련을 하겠습니다.

문제: 사용자 등급에 따른 '할인 대상자' 판별기를 만드세요.

메서드 설계: void checkDiscount(String grade, int couponCount) 메서드를 만드세요.

판별 조건: 아래 조건 중 **하나라도 참(||)**이면 "할인 적용 대상입니다"를 출력하고, 아니면 "일반 대상입니다"를 출력하세요.

grade가 "VIP"이다. (주의: 문자열 비교는 .equals()를 권장하지만, 일단 연습이니 편하게 짜보세요.)

couponCount가 5개보다 많다(> 5).

실행 예시:

checkDiscount("VIP", 2); -> 할인 대상

checkDiscount("NORMAL", 10); -> 할인 대상

checkDiscount("NORMAL", 3); -> 일반 대상*/

class DiscountObject {
    // 필드변수 만들필요없음 애초에 메소드만 돌리니까 

    void checkDiscount(String grade, int couponCount){
        if(grade.equals("VIP") || couponCount>5){
            System.out.println("할인 적용 대상입니다");

        }
        else{
            System.out.println("일반 대상입니다");
        }
    }
    
}
class Main {
    public static void main(String[] args) {
        DiscountObject dobj = new DiscountObject();
        dobj.checkDiscount("VIP",6);
    }
}
