자식 클래스 1: Developer (개발자) — Employee 상속

for문을 사용하여 배열에 있는 모든 직원의 work()를 실행시키고, 이름과 최종 월급을 출력하세요.

class Employee {
    String name;
    int baseSalary;

    void work(){
        System.out.println("열심히 일합니다");
    }
    int calculateSalary(){
            return baseSalary;
    }
}
class Developer extends Employee{
    @Override

    void work(){
        System.out.println("코드를 작성하고 버그를 잡습니다 ");
    }
     @Override

    int calculateSalary(){
         return baseSalary + 1000000;
    }
    
}
class Designer extends Employee {
    @Override

    void work(){
        System.out.println("웹사이트 디자인을 기획합니다");
    }
    @Override

    int calculateSalary(){
        return baseSalary + 500000;
        
    }
    
}
// The main method must be in a class named "Main".
class Main {
    public static void main(String[] args) {
        Employee d = new Developer();
        d.name = "Ben";
        d.baseSalary = 5000000;
        System.out.println(d.baseSalary);

        Employee dsn = new Designer();
        dsn.name = "Cindy";
        dsn.baseSalary = 3000000;


        Employee [] Employees = new Employee [2];
        Employees [0] = d;
        Employees [1] = dsn;

        int i;
        i= 0;
        for (i=0;i<2;i++){
            Employees [i].work();
            System.out.println("이름은"+Employees[i].name+"최종월급은"+Employees[i].calculateSalary());
        }
        
    }
}
