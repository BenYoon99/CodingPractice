문제 설명
머쓱이네 피자가게는 피자를 일곱 조각으로 잘라 줍니다. 피자를 나눠먹을 사람의 수 n이 주어질 때, 모든 사람이 피자를 한 조각 이상 먹기 위해 필요한 피자의 수를 return 하는 solution 함수를 완성해보세요.







class Solution {
    public int solution(int n) {
        int answer = 0;
         
        
        /* n 15  7이하니까   15 1 2 3 4 5 6 7 21 나머지가 6이하면 그냥 하면됨  22 3.xx
            15고 나머지 7이하면 멈추고 
            7로 나누고 나눈값 다운캐스팅
            해서 실수부 다버리고 정수형으로 뽑아내면됨 그럼 
            22 <- 7로나누면 3.xx 근데 정수형으로 실수부 날리면 3됨 그럼 3판필요함 아니지 
            무조건 반올림을 쳐야하네 
            30 7 2 나눈값에 플러스 1해주면되겟다 
            아
            21 7 0  3 
            23   2  4 
            
            */       
        int a = 0;
        
        if (n%7==0){
            a=n/7;
        }
        else if(n%7>0) {
            a=((int)(n/7)+1);
        }
        answer = a;        
        return answer;
    }
}
