import java.util.*;
import java.lang.*;
import java.io.*;

/*🔱 Mission 1: '상어의 탄생' (Constructor 기초)
객체가 만들어질 때 이름을 바로 지어주는 기본 중의 기본입니다.

조건:

Shark 클래스: String name 필드.

생성자: 이름을 인자로 받아 필드에 저장.

showName() 메서드: "내 이름은 [name] 상어다!" 출력.

Main: "경북대"라는 이름의 상어 객체를 생성하고 메서드 호출.*/

class Shark  {
    String name;
    Shark(String name){
        this.name=name;
    }
    void showName(){
        System.out.println("내 이름은 "+name+" 상어다!");
    }
}
class Main {
    public static void main(String[] args) {
        Shark shark = new Shark("경북대");      /* 여기서 궁금한건 내가 생성자 안파고 그냥 Shark shark = new Shark("경북대"); 하면 이름값안생기나? 
            예전에 그렇게햇다가 걍 이름 달고 태어난거같앗는데  ㄴㄴ 오답임
            애초에 그러면 오류나고 내가 그냥 Shark shark =new shark()로 파는것도 결국은 
            Shark의 규격을 따르는 참조변수 shark만들고 힙에 새로운 인스턴스를 위한 영역 만들어주고 new로 ㅇㅇ
            그리고 거기에 생성자로 인스턴스 생성해주는거 
            저건 아무것도없는 기본생성자이고 저경우 컴파일할떄 클래스내부에 생성자가없어도 자동으로 생성자 있는걸로 처리해줌
            */
        shark.showName();
    }
}
