#include <stdio.h>
/*🦈 [미션: 상어의 프로필 카드 출력하기]
노트북을 켜고, 아래 결과 화면과 똑같이 
나오도록 코드를 짜보세요. 단순히 글자만 치는 게 아니라, 반드시 변수를 생성해서 그 값을 출력 구멍(%d, %c)에 끼워 넣어야 합니다.

1. 조건 (Constraints)
year (정수형): 올해 연도 2026 저장

target (문자형): 목표 대학의 앞글자 K (경북대) 저장

days (정수형): D-Day 100 저장

출력 형식: 아래와 똑같이 줄 바꿈을 활용해서 출력할 것.*/
    
int main() {
    int year = 2026;
    char target = 'k';
    int days = 360;
    
    printf("올해는 : %d\n", year);
    printf("목표대학은 : %c\n", target);
    printf("남은 날은 :%d\n", days);
    
    
    
    return 0;
}
