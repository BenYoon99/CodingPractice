제미나이 제작 코드 분석 

#include <stdio.h>

int main() {
    int age = 28;             // 정수형 변수 age 선언해주고 그 상자안에 28넣어주고 
    double targetScore = 825.0; // 실수형 변수 ts 선언후 점수넣어주고
    char rank = 'S';          // 문자형 변수 rank 선언후 문자 하나 넣어주고 

    printf("나의 나이: %d세\n", age); // printf 로 출력 (내 나이 ~~~ %d <- 정수형 출력 구멍 , 구멍으로 출력시킬 정수형 변수);
    printf("CCNA 목표 점수: %.1f점\n", targetScore); // printf 로 출력 ( %.1f <- 실수 첫째짜리까지만 출력시키는 구멍 줄바꿈 );
    
    printf("나의 목표 등급: %c\n", rank); // %c 문자출력구멍 줄바꿈

    return 0; //모든 실행문 종료를 알림 
}
