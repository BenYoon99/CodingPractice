

import java.util.*;
import java.lang.*;
import java.io.*;


/*1. 클래스와 생성자 기초

문제: Student 클래스를 만드십시오. 필드는 name(이름), score(점수)를 가집니다. 생성자를 통해 두 필드를 초기화하고, 
    showInfo() 메서드로 "이름: OOO, 점수: OO"를 출력하는 코드를 짜십시오.*/

class Student {
    String name;
    int score;

    void showInfo(String name, int score){
        System.out.println("이름은: "+name+" 점수: "+score);
    }
    
}
class Main {
    public static void main(String[] args) {
        Student st = new Student();
        st.showInfo("홍길동",150);
            
    }
}

import java.util.*;
import java.lang.*;
import java.io.*;


/*1. 클래스와 생성자 기초

문제: Student 클래스를 만드십시오. 필드는 name(이름), score(점수)를 가집니다. 생성자를 통해 두 필드를 초기화하고, 
    showInfo() 메서드로 "이름: OOO, 점수: OO"를 출력하는 코드를 짜십시오.*/

class Student {
    String name;
    int score;

    Student(String name, int score){
        this.name=name;
        this.score=score;
    }
    void showInfo(String name, int score){
        System.out.println("이름은: "+name+" 점수: "+score);
    }
    
}
class Main {
    public static void main(String[] args) {
        Student st = new Student("홍영식",110);
        st.showInfo("홍길동",150);
            
    }
}
