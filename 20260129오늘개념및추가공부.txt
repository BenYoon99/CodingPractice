구성 = for ( 클래스 참조변수 : 배열명)
전 객실( 전 배열을 전부 ) 확인하며 거기있는 리모컨 하나씩 꺼내서  명령(c) 내려라.

**제미나이가 보여준 설명을 위한 코드  

// 1. 설계도 (Champion)
class Champion {
    String name;

    Champion(String name) { // 생성자: 태어날 때 이름을 지어줌
        this.name = name;
    }

    void 이동() {
        System.out.println(this.name + "이(가) 이동합니다.");
    }
}

public class Main {
    public static void main(String[] args) {
        // 2. 인스턴스(실제 객체) 생성
        Champion g1 = new Champion("가렌1");
        Champion g2 = new Champion("가렌2");
        Champion g3 = new Champion("가렌3");

        // 3. 리모컨 호텔(배열) 건설
        // Champion 타입의 리모컨 3개를 담을 수 있는 호텔을 짓고, 
        // 각 방에 g1, g2, g3 리모컨을 넣었습니다.
        Champion[] army = { g1, g2, g3 };

        // 4. 향상된 for문 (호텔 지배인의 순찰)
        System.out.println("=== 향상된 for문 시작 ===");
        
        for (Champion c : army) {
            // [해설]
            // army : 리모컨들이 담긴 호텔(배열)
            // :    : "호텔 1번 방부터 끝 방까지 순서대로 열어라"
            // Champion c : 각 방에서 꺼낸 리모컨을 잠시 쥐고 있을 지배인의 '손'
            
            c.이동(); // 지배인이 손에 쥔 리모컨(c)의 버튼을 누름!
        }
    }
}
