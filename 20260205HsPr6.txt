// 선생님 자체 제작문제 : 일반변수에 값 할당과정 설명

class Main {
  public static void main(String[] args) {
    // 설명
    int x;
        /* 1.메인메소드내부에 x라는 변수선언 
            2.이 변수는 정수형만 받을수있다 <-int의 의미
            3. 초기에 값을 넣어주지않았기에 쓰레기값인 0이들어감 ( 정수형이기에 0.0이건 딴거아니고 그냥 0 )*/
            
    // 설명
    x = 20;
      // 정수형타입만을 받는 변수에 20이라는값 집어넣음(저장) , 집어넣음과 동시에 초기에 존재하던 쓰레기값 0은 사라진다. 
  }
}
