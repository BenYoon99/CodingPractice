import java.util.*;
import java.lang.*;
import java.io.*;


/*🔱 Mission 2: '계산기 확장' (Overriding)
부모의 기능을 자식이 어떻게 '업그레이드' 하는지 확인하는 문제입니다.

조건:

Calculator 부모 클래스: int add(int n1, int n2) 메서드 (두 수의 합을 리턴).

ScienceCalculator 자식 클래스: add 메서드를 오버라이딩.

두 수의 합이 100을 넘으면 "와우! 큰 숫자입니다"를 출력하고 합을 리턴.

100 이하면 그냥 합만 리턴.

Main: 자식 클래스 객체를 생성해서 두 번 호출해 보세요. (합이 100 넘는 경우 / 안 넘는 경우)*/

class Calculator {
        public int add(int n1, int n2){
            int a = n1+n2;
            return a;
    }
}
class ScienceCalculator extends Calculator {
    @Override

    public int add(int n1, int n2){
        int b=n1+n2;
        if(b>100){
            System.out.println("와우! 큰숫자입니다");
            return b;
            
        }
        else{
            return b;
        }
    }
}
class Main {
    public static void main(String[] args) {
        Calculator sciCal = new ScienceCalculator();
        sciCal.add(65,50);
        
    }
}
