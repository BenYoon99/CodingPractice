import java.util.*;
import java.lang.*;
import java.io.*;

/*💻 실습 1: 인터페이스 구현과 부모 타입 참조 (동적 할당)
[문제]

RemoteControl 인터페이스를 만드세요. 메소드는 void turnOn(); 하나만 가집니다.

Electronic 클래스를 만드세요. 속성으로 String brand = "KNU";를 가집니다.

TV 클래스는 Electronic을 상속받고 RemoteControl을 구현합니다.

turnOn()을 오버라이딩하여 "TV가 켜졌습니다."를 출력하게 하세요.

Main에서 RemoteControl rc = new TV(); 로 동적 할당을 하세요.

[미션] rc를 이용해 TV를 켜고, 강제 형변환(Casting)을 사용해 brand 명을 출력하세요.*/

interface RemoteControl {
    void turnOn();
    
}
class Electronic {
    String brand ="KNU";
}
class TV  extends Electronic implements RemoteControl {
    public void turnOn(){
        System.out.println("TV가 켜졌습니다");
    }
}


class Main {
    public static void main(String[] args) {
        RemoteControl rc = new TV();
        rc.turnOn();
        System.out.println(((TV)rc).brand);
        
    }
}
