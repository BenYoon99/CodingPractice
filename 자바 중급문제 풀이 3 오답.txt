공통 필드: String name, int price

추상 메서드: void consume() (음료를 마실 때 각기 다른 메시지 출력)

abstract class Product {

    String name;
    int price;

    abstract void consume();

    }

class Coke extends Product {

    @Override

    void consume(){
        System.out.println("drinking coke ");
        
    }
}
class Tea extends Product {

    @Override

    void consume(){
        System.out.println("Drinking tea");
        
    }
}
class EnergyDrink extends Product {

    @Override

    void consume(){
        System.out.println("drinking Energydrink");
        
    }
}
