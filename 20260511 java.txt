class Main {
    public static void main(String[] args) {
        String str = "123";
        int num01 = Integer.parseint(str);
        Integer num02 = Integer.valueOf(str);
        System.out.println("num01:"+(num01+300));

        String str01 ="1010";
        int num03 = Integer.toBinaryString(str01);
        int num04 = Integer.parseint(str01,2);
        System.out.println("num03 :"+num03);
        System.out.println("num04 :"+num04);

        int a =13;
        String str02 = Integer.toString(a);
        String str03 = Integer.toBinaryString(a);
        System.out.println("str02 :"+str02);

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("숫자를 입력하세요";
        
        int num1 = sc.nextInt();

        try {

            int num2 =10/num1;
            System.out.println(num2);
        }
        catch (Exception e){
            System.out.println("올바른 숫자를 입력하세요");

            System.out.println(e.getMessage());
            System.out.println(e.printStackTrace());          
        }
        finally{
            System.out.println(" Try~ catch 블록의 외부 문장입니다");
        }
        
    }
}

package ch12; 

import java.io.File; 
import java.io.IOException; 

public class Example01 {
    public static void main(String[] args) {
        
        File file = new File("example01.txt");

        try {
            
            boolean success = file.createNewFile();

            
            if (success) {
                System.out.println("파일 생성에 성공하였습니다! : " + file.getName());
            } else {
                System.out.println("이미 동일한 이름의 파일이 존재합니다.");
            }
            
        } catch (IOException e) {
            
            System.out.println("오류가 발생하여 파일을 생성하지 못했습니다.");
            e.printStackTrace();
        }
    }
}

import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) {
        String sentence = " Java 92 , C 85 , C++ 81, Python 95 ";
        
        
        StringTokenizer st = new StringTokenizer(sentence, ", ");
        
        int sum = 0;
        int count = 0;
        
    System.out.println("--- 과목별 점수 내역 ---");
        
        while (st.hasMoreTokens()) {
            String subject = st.nextToken(); 
            
            if (st.hasMoreTokens()) {
                String scoreToken = st.nextToken(); 
                
                try {
                    int score = Integer.parseInt(scoreToken);
                    System.out.println(subject + " : " + score + "점");
                    
                    sum += score;
                    count++;
                } catch (NumberFormatException e) {
                    
                }
            }
        }
        
        System.out.println("----------------------");
        double average = (count > 0) ? (double) sum / count : 0;
        
        System.out.println("전체 합계: " + sum);
        System.out.println("전체 평균: " + average);
    }
}
