Java

class Counter {
    int count = 0;
}

public class HardStep2 {
    public static void main(String[] args) {
        Counter myCounter = new Counter(); // (ㅁ) 땅 파고 count=0
        
        reset(myCounter); // 메서드 호출 (주소값 복사해서 보냄)
        
        System.out.println(myCounter.count); // 최종 결과는?
    }

    public static void reset(Counter sub) {
        sub = new Counter(); // [여기!] sub가 갑자기 (ㅂ) 땅을 새로 파버림
        sub.count = 100;
    }
}

질문: myCounter.count는 0일까요, 100일까요?
