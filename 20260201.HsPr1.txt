public class MyPlan {
    // 1. 토익스코어라는 내 인스턴스변수값을 변동할수없게 자물쇠채우기 private
    private int toeicScore = 0;

    /* 2. 외부에서(클래스에서만 값바꿀수있엇는데 이거 사용하면 메인메소드에서 가능 ) 내 토익 점수를 바꾸고 싶을 때 쓰는 '창구' (Setter) 
                여기서 int score 파라미터에는 내가 이미 박은 int toeicscore가 들어감 */
                
    public void setToeicScore(int score) {
        if (score > 990) { // 이상한 값이 들어오면 필터링!
            System.out.println("990점이 만점임 ㅇㅇ, 오류!");
        } else {
            this.toeicScore = score;
            System.out.println("점수가 업데이트되었습니다.");
        }
    }

    // 3. 외부에서 내 토익 점수를 궁금해할 때 알려주는 '창구' (Getter)
    public int getToeicScore() {
        return this.toeicScore;
    }
}

// [1. 설계도 존: MyPlan 클래스]
class MyPlan {
    // 인스턴스 변수: 자물쇠(private)를 채운 내 소중한 데이터 금고 사용이유는 데이터 무결성을 위해 setter문 제외 바꿀수없게하기위해
    private int toeicScore = 0;

    // ★ 세터(Setter): 외부의 숫자를 내 금고로 옮겨주는 '검문소'
    public void setToeicScore(int score) { 
        // 여기서 (int score)는 990을 담아올 '빈 바구니'야. ( 빈바구니를 좀더 ㄱㅊ게 말하려면 파라미터 참조변수여기선 아님 ) 
        
        if (score > 990) { 
            System.out.println("오류: 990점이 만점입니다!"); 
        } else {
            // 핵심: 바구니(score)에 담긴 990을 진짜 금고(this.toeicScore)에 덮어쓰기!
            this.toeicScore = score; 
            System.out.println("로그: 데이터가 안전하게 저장되었습니다.");
        }
    }
	// 위처럼 세터문으로 틀을 만들어놓고 메인메소드에서 정보받을각 잡는걸 볼수있음

    // 겟터(Getter): 금고에 있는 값을 복사해서 밖으로 보여주는 '창구'
    public int getToeicScore() {
        return this.toeicScore;
    }
}

// [2. 실제 실행 존: 메인 메서드]
public class Main {
    public static void main(String[] args) {
        // 1. 설계도를 가지고 'myGoal'이라는 실제 객체(인스턴스)를 만듦
        MyPlan myGoal = new MyPlan();        //마이플랜의 규격따르는 마이골이라는 참조변수는 마이플랜을 따르는 인스턴스의 주솟값을 받는 리모컨임

        // 2. ★ 여기가 네가 묻던 '990을 박는 곳'이야!
        // 세터(setToeicScore)라는 창구에 990을 던져넣는 순간이지.
        myGoal.setToeicScore(990); 

        // 3. 겟터로 확인하기
        System.out.println("내 목표 점수: " + myGoal.getToeicScore());
    }
}
