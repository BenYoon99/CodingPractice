import java.util.*;

// The main method must be in a class named "Main".

class Person{
    String name;

    public void speak(){
        System.out.println("나의 이름은 "+ name + "입니다");
    }
}
class Singer extends Person{

    int age;

    @Override

    public void speak(){
        System.out.println("나의 이름은"+ name +"입니다 그리고 "+ age+"살입니다");
    }
}

class Main {
    public static void main(String[] args) {
        Person HelloPerson = new Singer();

        HelloPerson.name = "홍길동";
        ((Singer)HelloPerson).age = 23; // 참조변수에 리모콘 존재안하니까 다운캐스팅사용함 
        HelloPerson.speak();

    }

public class Main {
    public static void main(String[] args) {
        // 1. 2차원 배열 초기화 (4행 2열) 각성적 받고나서 
        double[][] scores = {
            {3.3, 3.5}, 
            {3.6, 3.7}, 
            {3.9, 4.1},
            {4.1, 4.3}  
        };

        double totalSum = 0;
        int totalCount = 0; 

        System.out.println("=== 학년별 평균 출력 (for문 활용) ===");
        
        //for
        for (int i = 0; i < scores.length; i++) {
            double yearSum = 0;
            for (int j = 0; j < scores[i].length; j++) {
                yearSum += scores[i][j]; // 해당 학년의 학기 점수 합산
                totalSum += scores[i][j]; // 전체 합계에도 누적
                totalCount++;
            }
            double yearAvg = yearSum / scores[i].length;
            System.out.println(String.format("%d학년 평균: %.2f", (i + 1), yearAvg)); // %d에 학년이니까 i+1 , %.2에 연평균점수 넣고
            
        }

        System.out.println("\n=== 전체 평균 출력 (for-each문 활용) ===");
        
        //for each 
        double sumForEach = 0;
        for (double[] yearArray : scores) {
            for (double semesterScore : yearArray) { // 그 안에서 점수를 하나씩 꺼냄
                sumForEach += semesterScore;
            }
        }
        
        double totalAvg = sumForEach / totalCount;
        System.out.println(String.format("전체 4학년 총 평균: %.2f", totalAvg));
    }
}
