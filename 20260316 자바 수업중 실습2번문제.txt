import java.util.*;
import java.lang.*;
import java.io.*;

// The main method must be in a class named "Main".


class Variable02 {
    static int classVar = 100;

    int instanceVar = 200;
    
}


class Main {
    public static void main(String[] args) {
        // 인스턴스 생성 
        Variable02 var1 = new Variable02();

        Variable02 var2 = new Variable02();

        var1.classVar = 110;

        var1.instanceVar = 220;

        System.out.println("var1정적변수:"+var1.classVar+"var1인스턴스변수:"+var1.instanceVar);
        
        
        
        
        System.out.println("var2정적변수:"+var2.classVar+"var2인스턴스변수:"+var2.instanceVar);
        
    }
}
