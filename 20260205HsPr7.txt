// 문제 : 레퍼런스변수에 값 할당과정 설명

class Main {
  public static void main(String[] args) {
    //설명  자동차라는 클래스의 규격을따르는 a자동차라는 레퍼런스변수 ( 리모컨 )생성 
    자동차 a자동차;
    
    // 설명 a자동차라는 레퍼런스변수는 새롭게 만든 자동차의 규격을따르는 인스턴스의 주솟값을 받아 그것을 조종할수있다 
    a자동차 = new 자동차();
    
    // 설명 a자동차라는 참조변수로 자동차라는 클래스내부에 메소드인 달리다();를 a자동차가 해당하는 인스턴스변수에게 지시(조종하는것)
    a자동차.달리다();

    // 모든 객체들은 서로를 평생 볼 수 없다.
    // 모든 객체들은 서로를 원격조종한다.
  }
}

// 자동차 설계도!, 절대 자동차가 아니다.!!!
class 자동차 {
  // 이 메서드(함수)의 2가지 용도
  // - 인스턴스를 제작할때 -  인스턴스가 실제로 작동되게하는 실제의 기능  
  // - 리모컨 (참조변수)를 제작할때- 참조변수가 가져올수있는 참조변수의 리모컨버튼
  void 달리다() {
    System.out.println("자동차가 달립니다.");
  }
}
