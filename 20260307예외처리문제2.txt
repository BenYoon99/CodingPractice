import java.util.*;
import java.lang.*;
import java.io.*;

/*🔱 다음 미션: [초급 2] 배열 인덱스 초과 (ArrayIndexOutOfBoundsException)
[문제]

크기가 2인 정수 배열을 만드십시오. (예: int[] arr = {10, 20};)

try 블록 안에서 존재하지 않는 인덱스인 **arr[5]**에 값을 넣거나 출력하려고 시도하십시오. (여기서 사고 유발!)

catch (ArrayIndexOutOfBoundsException e) 블록에서 사고를 잡고 **"상어님, 배열의 범위를 확인하십시오!"**라고 출력하십시오.

finally에서는 **"배열 점검 완료"**를 출력하십시오.*/


class Main {
    public static void main(String[] args) {
        int [] arr = {1, 2};

        try{
            arr[4] = 3;
            
        }
        catch(ArrayIndexOutOfBoundsException e){
            System.out.println("사고가 발생하였습니다 에러코드: "+e.getMessage()+"상어님, 배열의 범위를 확인하십시오!");
            
        }
        finally{
            System.out.println("배열 점검 완료");
        }
    }
}
