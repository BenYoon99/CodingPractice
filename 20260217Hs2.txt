import java.util.ArrayList;

interface Attackable {
    void attack();
}

// Warrior 클래스가 Attackable을 구현하게 합니다.
// 이렇게 하면 '전사'는 '무조건 공격 가능한 존재'가 됩니다.
class Warrior implements Attackable {
    String name;
    int power;

    void showStatus() {
        System.out.println("이름: " + name + " | 공격력: " + power);
    }

    // 부모 클래스에서 기본 attack을 정의 (자식이 오버라이딩 할 예정)
    @Override
    public void attack() {
        System.out.println("기본 공격");
    }
}

class Swordsman extends Warrior {
    @Override
    public void attack() { // 인터페이스 구현 시 반드시 public!
        System.out.println("검으로 베기!");
    }
}

class Archer extends Warrior {
    @Override
    public void attack() {
        System.out.println("활로 쏘기!");
    }
}

import java.util.ArrayList;

// 1. 규격 선언: 모든 공격 가능한 객체는 attack()을 가져야 한다.
interface Attackable {
    void attack();
}

// 2. 추상 클래스: 공통 속성(name, power)을 관리하되, attack의 내용은 자식에게 떠넘긴다.
// implements Attackable을 했지만, 내용을 채우지 않고 abstract로 선언함.
abstract class Warrior implements Attackable {
    String name;
    int power;

    // 공통 메서드 (모든 전사가 동일하게 사용)
    void showStatus() {
        System.out.println("이름: " + name + " | 공격력: " + power);
    }

    // 추상 메서드: "Warrior를 상속받는 놈은 무조건 attack을 오버라이딩해라!" (강제성)
    @Override
    public abstract void attack(); 
}

// 3. 자식 클래스: 부모가 떠넘긴 attack()을 반드시 구현해야 함.
class Swordsman extends Warrior {
    @Override
    public void attack() {
        System.out.println("🗡️ 검으로 베기! (150% 데미지)");
    }
}

class Archer extends Warrior {
    @Override
    public void attack() {
        System.out.println("🏹 활로 쏘기! (원거리 저격)");
    }
}

// 4. 메인 클래스
class Main {
    public static void main(String[] args) {
        // Warrior라는 안경(부모 타입)으로 자식들을 바라봄 -> 다형성
        Warrior s = new Swordsman();
        s.name = "상어검사";
        s.power = 150;

        Warrior a = new Archer();
        a.name = "상어궁수";
        a.power = 130;

        // Warrior 타입 리스트에 담기 (Warrior는 Attackable이기도 하므로 OK)
        ArrayList<Warrior> champions = new ArrayList<>();
        champions.add(s);
        champions.add(a);

        System.out.println("=== 전사 군단 상태 보고 및 공격 시작 ===");
        
        // 향상된 for문
        for (Warrior w : champions) {
            w.showStatus(); // Warrior의 능력 사용
            w.attack();     // 자식이 재정의한(오버라이딩) 능력 사용
            System.out.println("---------------------------");
        }
    }
}
